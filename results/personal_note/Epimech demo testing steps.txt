  =====================================================================
STRONGER dt_NaN suppression (module-filtering logger)
=====================================================================
  Type this whole block at the julia> prompt, once per session,
  before running any include(...). It replaces the simple
  disable_logging approach from before -- use this one instead.

  julia> using Logging

  julia> struct QuietSciMLLogger <: Logging.AbstractLogger
             inner::Logging.AbstractLogger
         end
         Logging.min_enabled_level(l::QuietSciMLLogger) = Logging.min_enabled_level(l.inner)
         Logging.catch_exceptions(l::QuietSciMLLogger) = Logging.catch_exceptions(l.inner)
         function Logging.shouldlog(l::QuietSciMLLogger, level, _module, group, id)
             modname = _module === nothing ? "" : string(_module)
             if occursin("OrdinaryDiffEq", modname) || occursin("SciMLBase", modname)
                 return false
             end
             return Logging.shouldlog(l.inner, level, _module, group, id)
         end
         Logging.handle_message(l::QuietSciMLLogger, args...; kwargs...) =
             Logging.handle_message(l.inner, args...; kwargs...)

  julia> global_logger(QuietSciMLLogger(current_logger()))

  This filters out any log message whose SOURCE MODULE name contains
  "OrdinaryDiffEq" or "SciMLBase" -- exactly the two modules shown in
  the warning stack traces -- while every other package's warnings
  (and all genuine errors) still print normally.

  To go back to normal logging in the same session:
  julia> global_logger(ConsoleLogger())
